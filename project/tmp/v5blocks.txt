function dayStops(ls, fromName, toName, dep, sunset) {
  const out = []; let t = dep, driveH = 0, stopMin = 0, km = 0, pend = { h: 0, km: 0, roads: [] };
  out.push({ time: clock(t), kind: 'go', label: `Leave ${fromName}`, note: '' });
  const road = r => { if (!pend.roads.includes(r)) pend.roads.push(r); };
  const flush = () => { if (pend.h > 0.02) out.push({ isDrive: true, kind: 'drive', time: '', label: `${dur(pend.h)} drive`, note: `${fmtN(pend.km < 20 ? pend.km : Math.round(pend.km / 5) * 5)} km · ${pend.roads.join(', ')}` });
    t += pend.h; driveH += pend.h; km += pend.km; pend = { h: 0, km: 0, roads: [] }; };
  ls.forEach(l => { const sp = l.km / Math.max(l.h, 0.01); let done = 0; road(l.road);
    stopsOn(l).forEach(([at, kind, label, note]) => { pend.h += (at - done) / sp; pend.km += at - done; done = at; flush(); const m = STOP_DUR[kind] || 15;

    pend.h += (l.km - done) / sp; pend.km += l.km - done; });
  flush();
  const late = t > sunset;
  out.push({ time: clock(t), kind: 'bed', label: `Arrive ${toName}`, note: late ? `After sunset (about ${clock(sunset)}). Leave earlier if you can.` : `Before sunset (about ${clock(sunset)})` });
  return { items: out, arrive: t, driveH, stopMin, km, late };
}

function trip(key, origin, cap = MAX_DRIVE_H, month = 7) {

  let n = start.n;
